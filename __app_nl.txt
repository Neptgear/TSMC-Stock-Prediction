 110: def build_charts(feat_df: pd.DataFrame) -> Dict[str, Any]:
 111:     """Prepare chart data consumable by Plotly in the frontend."""
 112:     import plotly.graph_objs as go
 113:     from plotly.utils import PlotlyJSONEncoder
 114: 
 115:     charts: Dict[str, Any] = {}
 116:     idx = feat_df.index
 117: 
 118:     # Price + MAs
 119:     price_traces = [
 120:         go.Scatter(x=idx, y=feat_df["Close"], name="Close", mode="lines")
 121:     ]
 122:     for ma_col, name in [("ma5", "MA5"), ("ma10", "MA10"), ("ma20", "MA20")]:
 123:         if ma_col in feat_df.columns:
 124:             price_traces.append(go.Scatter(x=idx, y=feat_df[ma_col], name=name, mode="lines"))
 125:     price_layout = go.Layout(title="Price & Moving Averages", legend={"orientation": "h"})
 126:     price_fig = go.Figure(data=price_traces, layout=price_layout)
 127:     charts["price"] = json.loads(price_fig.to_json())
 128: 
 129:     # Volume
 130:     if "Volume" in feat_df.columns:
 131:         vol_fig = go.Figure(data=[go.Bar(x=idx, y=feat_df["Volume"], name="Volume")], layout=go.Layout(title="Volume"))
 132:         charts["volume"] = json.loads(vol_fig.to_json())
 133: 
 134:     # RSI
 135:     if "rsi14" in feat_df.columns:
 136:         rsi_fig = go.Figure(
 137:             data=[go.Scatter(x=idx, y=feat_df["rsi14"], name="RSI14", mode="lines")],
 138:             layout=go.Layout(title="RSI(14)", yaxis={"range": [0, 100]}),
 139:         )
 140:         charts["rsi"] = json.loads(rsi_fig.to_json())
 141: 
 142:     # MACD
 143:     if set(["macd", "macd_signal", "macd_hist"]).issubset(set(feat_df.columns)):
 144:         macd_fig = go.Figure(
 145:             data=[
 146:                 go.Scatter(x=idx, y=feat_df["macd"], name="MACD", mode="lines"),
 147:                 go.Scatter(x=idx, y=feat_df["macd_signal"], name="Signal", mode="lines"),
 148:                 go.Bar(x=idx, y=feat_df["macd_hist"], name="Hist"),
 149:             ],
 150:             layout=go.Layout(title="MACD"),
 151:         )
 152:         charts["macd"] = json.loads(macd_fig.to_json())
 153: 
 154:     return charts
